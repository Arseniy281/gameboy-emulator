cmake_minimum_required(VERSION 3.20)
project(GameBoyEmulator CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

enable_testing()

add_subdirectory(src/cpu)
add_subdirectory(src/ppu)
add_subdirectory(tests/cpu)
add_subdirectory(src/doctor)

# SDL2 main executable (optional if SDL2 is available)
find_package(PkgConfig QUIET)
if (PkgConfig_FOUND)
    pkg_check_modules(SDL2 IMPORTED_TARGET sdl2)
endif()

set(GB_HAVE_SDL FALSE)
if (TARGET PkgConfig::SDL2)
    set(GB_HAVE_SDL TRUE)
else()
    find_package(SDL2 QUIET)
    if (SDL2_FOUND)
        set(GB_HAVE_SDL TRUE)
    endif()
endif()

if (GB_HAVE_SDL)
    add_executable(gb_main src/main.cpp)
    target_include_directories(gb_main PRIVATE src/cpu src/ppu)
    target_link_libraries(gb_main PRIVATE gb_cpu gb_ppu)
    if (TARGET PkgConfig::SDL2)
        target_link_libraries(gb_main PRIVATE PkgConfig::SDL2)
    else()
        target_include_directories(gb_main PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(gb_main PRIVATE ${SDL2_LIBRARIES})
    endif()
else()
    message(STATUS "SDL2 not found; skipping gb_main target")
endif()
